- keyboard:f: FAVORITE_ADD, FAVORITE_REMOVE

### UNCAT
- keyboard:/: FILTER, SEARCH
- keyboard:DOWN: DOWN, TEXT.DOWN
- keyboard:ESC: QUIT, TEXT.QUIT

- keyboard:NPAGE: LEVEL_DOWN, PAGE_DOWN, TEXT.PAGE_DOWN
- keyboard:NUMPAD_6: RIGHT, TEXT.CONFIRM
- keyboard:PPAGE: LEVEL_UP, PAGE_UP, TEXT.PAGE_UP
- keyboard:RETURN: CONFIRM, TEXT.CONFIRM
- keyboard:RIGHT: RIGHT, TEXT.RIGHT
- keyboard:TAB: EDITMAP_TAB, NEXT_TAB
- keyboard:UP: TEXT.UP, UP
